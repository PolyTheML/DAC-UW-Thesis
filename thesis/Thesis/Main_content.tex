% Captions carry their own manual number ("Figure 7.", "Table 9") to match
% the in-text references, so suppress LaTeX's auto label in the caption
% (labelformat=empty) AND the auto counter number in the LoF/LoT entries
% (listformat=empty) --- otherwise every list line shows two disagreeing
% numbers ("5.1  Figure 7. ..."). \caption still writes the entry + page.
\captionsetup[figure]{labelformat=empty, listformat=empty}
\captionsetup[table]{labelformat=empty, listformat=empty}
% Pandoc emits \captionof{table}{...}\label{tab:N} before each longtable. With
% labelformat=empty the caption establishes no counter, so the trailing \label
% (and hyperref's \theHnone) reference an undefined 'none' counter -> hard
% "No counter 'none' defined" error at every table. No \ref/\autoref targets
% these labels (all in-text numbers are manual text), so a dummy counter just
% silences the error; the resulting anchors are unused and harmless.
\newcounter{none}
\providecommand\theHnone{none.\arabic{none}}

% ── Numbered-chapter heading: "Chapter I. Title" ─────────────────────────────
% The Roman numeral is shown in the heading text only (via \Roman{chapter}); we
% do NOT redefine \thechapter, so section numbers stay arabic (4.1, 4.2, …) and
% every manual "Section 4.x" reference remains valid. \chapter* (front matter,
% TOC, bibliography, appendices) uses \@makeschapterhead and is left untouched.
\makeatletter
\def\@makechapterhead#1{%
  \vspace*{30\p@}%
  {\parindent \z@ \centering \normalfont
    \ifnum \c@secnumdepth >\m@ne
        \LARGE\bfseries \@chapapp\space \Roman{chapter}.\space #1%
    \else
        \LARGE\bfseries #1%
    \fi
    \par\nobreak \vskip 30\p@ }}
\makeatother

% Manual figure numbering: captions already carry "Figure N." (matching in-text
% references), so labelformat=empty (see Settings) hides LaTeX's own label while
% \caption still registers the entry in the List of Figures. The optional first
% argument is a SHORT title for the List of Figures ("Figure N. <one line>", no
% trailing period); without it the full caption is used as the list entry.
\newcommand{\fg}[4][]{
    \begin{figure}[H]
        \centering
        \includegraphics[width=#2]{#4}
        \ifblank{#1}{\caption{#3}}{\caption[#1]{#3}}
        \label{fig:#4}
    \end{figure}
}

            }%
        }

        \label{fig:#3}

}
